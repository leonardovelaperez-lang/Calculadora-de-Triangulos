  <script>

    const caseSelect = document.getElementById('caseSelect');

    const inputsDiv = document.getElementById('inputs');

    function renderInputs() {

      const value = caseSelect.value;

      if (value === '1') {

        inputsDiv.innerHTML = `

          <label>Lado a</label>

          <input type="number" id="a" placeholder="Ingresa el lado a">

          <label>Ángulo X</label>

          <input type="number" id="X" placeholder="Ingresa el ángulo X">

          <label>Ángulo W</label>

          <input type="number" id="W" placeholder="Ingresa el ángulo W">

        `;

      }

      if (value === '2') {

        inputsDiv.innerHTML = `

          <label>Lado a</label>

          <input type="number" id="a" placeholder="Ingresa el lado a">

          <label>Lado b</label>

          <input type="number" id="b" placeholder="Ingresa el lado b">

          <label>Ángulo Z</label>

          <input type="number" id="Z" placeholder="Ingresa el ángulo Z">

        `;

      }

      if (value === '3') {

        inputsDiv.innerHTML = `

          <label>Lado a</label>

          <input type="number" id="a" placeholder="Ingresa el lado a">

          <label>Lado b</label>

          <input type="number" id="b" placeholder="Ingresa el lado b">

          <label>Lado c</label>

          <input type="number" id="c" placeholder="Ingresa el lado c">

        `;

      }

    }

    renderInputs();

    caseSelect.addEventListener('change', renderInputs);

    function toRadians(degrees) {

      return degrees * Math.PI / 180;

    }

    function toDegrees(radians) {

      return radians * 180 / Math.PI;

    }

    function calculateTriangle() {

      const option = caseSelect.value;

      let a = 0;

      let b = 0;

      let c = 0;

      let X = 0;

      let W = 0;

      let Z = 0;

      if (option === '1') {

        a = parseFloat(document.getElementById('a').value);

        X = parseFloat(document.getElementById('X').value);

        W = parseFloat(document.getElementById('W').value);

        Z = 180 - (X + W);

        b = (a * Math.sin(toRadians(W))) / Math.sin(toRadians(X));

        c = (a * Math.sin(toRadians(Z))) / Math.sin(toRadians(X));

      }

      if (option === '2') {

        a = parseFloat(document.getElementById('a').value);

        b = parseFloat(document.getElementById('b').value);

        Z = parseFloat(document.getElementById('Z').value);

        c = Math.sqrt(a**2 + b**2 - 2 * a * b * Math.cos(toRadians(Z)));

        X = toDegrees(Math.acos((b**2 + c**2 - a**2) / (2 * b * c)));

        W = 180 - (X + Z);

      }

      if (option === '3') {

        a = parseFloat(document.getElementById('a').value);

        b = parseFloat(document.getElementById('b').value);

        c = parseFloat(document.getElementById('c').value);

        X = toDegrees(Math.acos((b**2 + c**2 - a**2) / (2 * b * c)));

        W = toDegrees(Math.acos((a**2 + c**2 - b**2) / (2 * a * c)));

        Z = 180 - (X + W);

      }

      let type = '';

      if (Math.abs(a - b) < 0.01 && Math.abs(b - c) < 0.01) {

        type = 'Equilátero';

      } else if (

        Math.abs(a - b) < 0.01 ||

        Math.abs(b - c) < 0.01 ||

        Math.abs(a - c) < 0.01

      ) {

        type = 'Isósceles';

      } else {

        type = 'Escaleno';

      }

      document.getElementById('results').style.display = 'block';

      document.getElementById('resultsContent').innerHTML = `

        <div class="result-item">Lado a: ${a.toFixed(2)}</div>

        <div class="result-item">Lado b: ${b.toFixed(2)}</div>

        <div class="result-item">Lado c: ${c.toFixed(2)}</div>

        <div class="result-item">Ángulo X: ${X.toFixed(2)}°</div>

        <div class="result-item">Ángulo W: ${W.toFixed(2)}°</div>

        <div class="result-item">Ángulo Z: ${Z.toFixed(2)}°</div>

        <div class="triangle-type">Tipo de triángulo: ${type}</div>

      `;

    }
